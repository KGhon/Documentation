[Callable,#4EC9B0]
[Helpers,#4EC9B0]
[UndoRedo,#4EC9B0]
[MergeMode,#B5CEA8]
[Ends,#D4D4D4]
[Export] private KDTExtensionInspectableDataBase _inspectableData;

public override void _GuiInput(InputEvent input)
{
	base._GuiInput(input);
	if (input is InputEventMouseButton mouseButton)
	{
		if (GetGlobalRect().HasPoint(GetGlobalMousePosition()))
		{
			if (mouseButton.Pressed && mouseButton.ButtonIndex == MouseButton.Right)
			{
				if (_inspectableData != null)
				{
					_inspectableData = _inspectableData.Duplicate(true) as KDTExtensionInspectableDataBase;
					IInspectReceiver extensionPanel = Helpers.FindParentByType<IInspectReceiver>(this);
					extensionPanel.Inspect(this, _inspectableData);
				}
			}
		}
	}
}
