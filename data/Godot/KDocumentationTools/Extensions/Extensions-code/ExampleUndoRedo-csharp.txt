[Callable,#4EC9B0]
[Helpers,#4EC9B0]
[UndoRedo,#4EC9B0]
[MergeMode,#B5CEA8]
[Disable,#D4D4D4]
private void ToggleComplete(bool isComplete)
{
	Callable doMethod;
	if (isComplete)
	{
		doMethod = Callable.From(() =>
		{
			_completeInput.ButtonPressed = true;
			if (!_markedInput.ButtonPressed)
			{
				_background.SelfModulate = Colors.Green;
			}
		});
	}
	else
	{
		doMethod = Callable.From(() =>
		{
			_completeInput.ButtonPressed = false;
			if (!_markedInput.ButtonPressed)
			{
				_background.SelfModulate = Colors.White;
			}
			else
			{
				_background.SelfModulate = Colors.Red;
			}
		});
	}
	doMethod.Call();

	if (undoRedo != null)
	{
		Callable undoMethod;
		if (isComplete)
		{
			undoMethod = Callable.From(() =>
			{
				_completeInput.ButtonPressed = false;
				if (!_markedInput.ButtonPressed)
				{
					_background.SelfModulate = Colors.White;
				}
				else
				{
					_background.SelfModulate = Colors.Red;
				}
			});
		}
		else
		{
			doMethod = Callable.From(() =>
			{
				_completeInput.ButtonPressed = true;
				if (!_markedInput.ButtonPressed)
				{
					_background.SelfModulate = Colors.Green;
				}
			});
		}

		Helpers.CommitUndoRedoAction(undoRedo, UndoRedo.MergeMode.Disable, itemName + ": Toggle complete", doMethod, undoMethod);
	}
}
